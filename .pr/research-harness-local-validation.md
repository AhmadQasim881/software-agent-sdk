## Repeatable operator sequence

Run from the repository root with the development dependencies already installed.
This sequence invokes the existing CLI's `main(argv)` entry point so the launcher
and cleanup share the owning Python process. That is needed in execution
services which reap background children when an individual shell invocation
ends. It preserves the existing argument parser, lifecycle, API assertions,
redaction and evidence writer. It is an operator recipe, not an agent tool.

`LITELLM_LOCAL_MODEL_COST_MAP=True` prevents an unrelated import-time remote
pricing-map lookup. No provider credentials are passed to the server.

```sh
.venv/bin/python - <<'PY'
import contextlib
import importlib.util
import io
import json
import os
from pathlib import Path
import subprocess
import sys
import tempfile

repo = Path.cwd()
os.environ['OPENHANDS_SUPPRESS_BANNER'] = '1'
os.environ['LITELLM_LOCAL_MODEL_COST_MAP'] = 'True'
root = Path(tempfile.mkdtemp(prefix='research-acceptance-'))
os.environ['AGENT_SERVER_VERIFY_HOME'] = str(root)
script = repo / '.agents/skills/verify-agent-server/scripts/control_agent_server.py'
spec = importlib.util.spec_from_file_location('acceptance_server_cli', script)
cli = importlib.util.module_from_spec(spec)
sys.modules[spec.name] = cli
spec.loader.exec_module(cli)
sha = subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip()
run = None
observations = {}

def invoke(label, argv, require_success=True):
    buffer = io.StringIO()
    with contextlib.redirect_stdout(buffer):
        code = cli.main(argv)
    payload = json.loads(buffer.getvalue())
    observations[label] = {'exit_code': code, 'result': payload}
    print(json.dumps({'step': label, **observations[label]}), flush=True)
    if require_success and code != 0:
        raise RuntimeError(f'{label} failed with exit {code}')
    return payload

try:
    launched = invoke('launch', ['launch', '--new', '--name', 'research-harness-readiness',
        '--env', 'OPENHANDS_SUPPRESS_BANNER=1',
        '--env', 'LITELLM_LOCAL_MODEL_COST_MAP=True', '--timeout', '60'])
    run = Path(launched['run'])
    invoke('doctor', ['doctor', '--run', str(run)])
    probes = [
        ('F01.ready', '/ready', ['--auth', 'none', '--check', 'status', 'eq', 'ready'],
         'HTTP 200; status=ready'),
        ('F01.server-info', '/server_info', ['--auth', 'none', '--check', 'title', 'eq',
         'OpenHands Agent Server', '--check', 'build_git_sha', 'eq', sha],
         f'HTTP 200; OpenHands Agent Server; build_git_sha={sha}'),
        ('F04.count', '/api/conversations/count', ['--check', '.', 'eq', '0'],
         'HTTP 200; count=0; search items empty'),
    ]
    for feature, endpoint, checks, expected in probes:
        invoke(feature, ['api', 'GET', endpoint, '--run', str(run), '--expect', '200',
            *checks, '--save', f'{feature}/acceptance'])
        if feature == 'F04.count':
            invoke('F04.count.search', ['api', 'GET', '/api/conversations/search',
                '--run', str(run), '--expect', '200', '--query', 'limit=100',
                '--check', 'items', 'len-eq', '0', '--save', 'F04.count/search'])
        evidence_args = ['--artifact', f'evidence/{feature}/acceptance.json']
        if feature == 'F04.count':
            evidence_args += ['--artifact', 'evidence/F04.count/search.json',
                '--note', 'Fresh empty catalog only; no model-backed conversation created.']
        invoke(f'{feature}.ledger', ['evidence', 'add', '--run', str(run), '--feature', feature,
            '--result', 'pass', '--entry', f'GET {endpoint}', '--expected', expected,
            '--actual', expected, *evidence_args])
    invoke('evidence-report', ['evidence', 'report', '--run', str(run),
        '--out', str(run / 'evidence' / 'acceptance-report.md')])
finally:
    if run is None:
        candidates = list(root.glob('*/run.json'))
        if len(candidates) == 1:
            run = candidates[0].parent
    if run is not None:
        stopped = invoke('stop', ['stop', '--run', str(run), '--purge-private'], False)
        private_dirs = ['private', 'home', 'server', 'tmux', 'worktrees', 'fixtures']
        cleanup = {
            'port_closed': stopped.get('port_closed'),
            'private_state_removed': all(not (run / name).exists() for name in private_dirs),
        }
        observations['cleanup'] = cleanup
        evidence = run / 'evidence' / 'acceptance-observations.json'
        evidence.write_text(json.dumps(observations, indent=2) + '\n')
        print(json.dumps({'evidence': str(evidence), 'cleanup': cleanup}), flush=True)
        if not stopped.get('ok') or not all(cleanup.values()):
            raise RuntimeError('Cleanup verification failed')
PY
```
